📋 Step 1: Run SQL in Supabase SQL Editor

Copy and paste this script directly into your Supabase SQL Editor to create the tables and secure Row Level Security (RLS) policies: 

-- 1. Create responses table
CREATE TABLE IF NOT EXISTS public.responses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  visitor_name TEXT NOT NULL,
  response_type TEXT NOT NULL DEFAULT 'YES',
  date_activity TEXT,
  food_choice TEXT,
  drink_choice TEXT,
  outfit_choice TEXT,
  preferred_date TEXT,
  personal_message TEXT,
  quiz_summary TEXT,
  is_read BOOLEAN DEFAULT false,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Enable Row Level Security (RLS)
ALTER TABLE public.responses ENABLE ROW LEVEL SECURITY;

-- 3. Security Policy: Visitors can insert responses
CREATE POLICY "Allow anonymous visitors to insert responses"
  ON public.responses FOR INSERT
  WITH CHECK (true);

-- 4. Security Policy: Owner can read responses
CREATE POLICY "Allow authenticated owner to read responses"
  ON public.responses FOR SELECT
  TO authenticated
  USING (true);

-- 5. Security Policy: Owner can update responses (e.g. mark as read)
CREATE POLICY "Allow authenticated owner to update responses"
  ON public.responses FOR UPDATE
  TO authenticated
  USING (true);

-- 6. Security Policy: Owner can delete responses
CREATE POLICY "Allow authenticated owner to delete responses"
  ON public.responses FOR DELETE
  TO authenticated
  USING (true);

-- 7. Create website_settings table for owner customizations
CREATE TABLE IF NOT EXISTS public.website_settings (
  id TEXT PRIMARY KEY DEFAULT 'default',
  owner_id UUID REFERENCES auth.users(id),
  website_title TEXT DEFAULT 'CAN WE GO ON A DATE? 💗',
  invitation_heading TEXT DEFAULT 'CAN WE GO ON A DATE?',
  invitation_subtitle TEXT DEFAULT 'I''ve been wanting to ask you something special. Sooo... 🥺',
  recipient_name TEXT DEFAULT 'My Favorite Person',
  sender_name TEXT DEFAULT 'Your Secret Admirer',
  love_letter TEXT,
  color_theme TEXT DEFAULT 'pink',
  customization_data JSONB,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

ALTER TABLE public.website_settings ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public to read website_settings"
  ON public.website_settings FOR SELECT
  USING (true);

CREATE POLICY "Allow authenticated owner to modify website_settings"
  ON public.website_settings FOR ALL
  TO authenticated
  USING (true)
  WITH CHECK (true);

🚀 Step 2: Deploying to GitHub Pages

  1. Upload index.html, style.css, and script.js to your GitHub repository root.
  2. In your GitHub repo, go to Settings → Pages.
  3. Under Source, select the main branch and folder / (root), then click Save.
  4. Your website will be live at https://<your-username>.github.io/<repo-name>/!
  5. Open the site, click Owner in the header, input your Supabase credentials once, and your invitation is fully live and connected!
